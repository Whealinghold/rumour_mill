type SourceType int

const (
	SourceWitnessed SourceType = iota
	SourceHeard
	SourceInferred
	SourceOfficial
)

func sourceStrength(s SourceType) int {
	switch s {
	case SourceWitnessed:
		return 3
	case SourceOfficial:
		return 3
	case SourceHeard:
		return 2
	case SourceInferred:
		return 1
	default:
		return 0
	}
}

type KnowledgeFact struct {
	ID         string
	Type       KnowledgeType
	Subject    shared.EntityID
	Object     interface{}
	Confidence float64
	Source     SourceType
	Timestamp  shared.GameTime
	Expiry     *shared.GameTime
	Secret     bool
}

func MakeFactID(eventType EventType, source shared.EntityID, subject shared.EntityID, timestamp shared.GameTime) string {
	return fmt.Sprintf("%s_%s_%s_%04d-%02d-%02dT%02d:%02d",
		eventType, source, subject,
		timestamp.Year, timestamp.Month, timestamp.Day,
		timestamp.Hour, timestamp.Minute)
}

// MakeKnowledgeFactID returns a stable fact ID that includes the knowledge type so
// multiple facts from the same event (e.g. witnessed_killing and entity_is_hostile) do not collide.
// objectKey is an optional discriminator for facts that store payload-specific data (e.g. victim ID).
func MakeKnowledgeFactID(factType KnowledgeType, eventType EventType, source shared.EntityID, holder shared.EntityID, timestamp shared.GameTime, objectKey string) string {
	id := string(factType) + ":" + MakeFactID(eventType, source, holder, timestamp)
	if objectKey != "" {
		id += ":" + objectKey
	}
	return id
}
